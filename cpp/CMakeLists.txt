cmake_minimum_required(VERSION 3.20)
project(phantom_cpp_foundation VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)

add_library(phantom_foundation SHARED src/api.cpp src/core.cpp src/runtime.cpp)
target_compile_features(phantom_foundation PUBLIC cxx_std_17)
target_include_directories(phantom_foundation PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(phantom_foundation PRIVATE PHANTOM_BUILD_DLL)
target_link_libraries(phantom_foundation PRIVATE Threads::Threads)
set_target_properties(phantom_foundation PROPERTIES OUTPUT_NAME phantom-internal-foundation)
if(WIN32)
    target_link_libraries(phantom_foundation PRIVATE user32 gdi32)
endif()
if(MSVC)
    target_compile_options(phantom_foundation PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(phantom_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(BUILD_TESTING)
    add_executable(phantom_core_tests tests/core_tests.cpp)
    target_link_libraries(phantom_core_tests PRIVATE phantom_foundation Threads::Threads)
    add_test(NAME cpp_core COMMAND phantom_core_tests)
    # Compile the C++ game core into this test executable. The DLL intentionally
    # exports only its stable C ABI on Windows, not C++ symbols.
    add_executable(phantom_game_core_tests tests/game_core_tests.cpp src/core.cpp)
    target_include_directories(phantom_game_core_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_features(phantom_game_core_tests PRIVATE cxx_std_17)
    target_link_libraries(phantom_game_core_tests PRIVATE Threads::Threads)
    add_test(NAME cpp_game_core COMMAND phantom_game_core_tests)
    if(WIN32)
        if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
            message(FATAL_ERROR "The Windows DLL target requires an x64 toolchain.")
        endif()
        add_executable(phantom_dll_smoke tests/dll_smoke.cpp)
        target_compile_features(phantom_dll_smoke PRIVATE cxx_std_17)
        target_include_directories(phantom_dll_smoke PRIVATE include)
        add_test(NAME windows_load_unload COMMAND phantom_dll_smoke $<TARGET_FILE:phantom_foundation>)
    endif()
endif()
